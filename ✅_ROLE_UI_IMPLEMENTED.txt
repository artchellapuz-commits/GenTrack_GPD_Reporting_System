========================================
  USAGE IN COMPONENTS
========================================

<template>
  <!-- Show only for operators+ -->
  <button v-if="canUpload">
    Upload Data
  </button>

  <!-- Show only for managers+ -->
  <button v-if="canApprove">
    Approve
  </button>

  <!-- Show only for admins -->
  <div v-if="isAdmin">
    Admin Controls
  </div>
</template>

<script>
import { 
  canUploadData, 
  canApproveData, 
  isAdmin 
} from '@/utils/auth';

export default {
  data() {
    return {
      canUpload: canUploadData(),
      canApprove: canApproveData(),
      isAdmin: isAdmin()
    };
  }
};
</script>
